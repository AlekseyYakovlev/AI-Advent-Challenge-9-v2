    status: KbStatus (SAEnum values_callable, default QUEUED); error: str | None;
    strategy: KbStrategy (SAEnum values_callable); chunk_size: int; chunk_overlap: int;

    file_count: int = 0; chunk_count: int = 0; done_chunks: int = 0; total_chunks: int = 0;
    phase: str | None (one of "loading_model", "parsing", "embedding", or None);
    created_at / updated_at: datetime (datetime.now(timezone.utc))
- class KbDocument(SQLModel, table=True)  -> table "kbdocument"
    id PK; kb_id: int (FK knowledgebase.id CASCADE, index); filename: str; sha256: str (64);
    size_bytes: int; page_count: int | None; stored_name: str (server-generated, relative to uploads dir)
    __table_args__ = (UniqueConstraint("kb_id", "sha256", name="uq_kbdocument_kb_sha256"),)
- class KbChunk(SQLModel, table=True)  -> table "kbchunk"   (KbChunk.id IS the FAISS id)
    id PK; kb_id: int (FK knowledgebase.id CASCADE, index); document_id: int (FK kbdocument.id CASCADE, index);
    chunk_index: int; chunk_id: str  (format "{document_id}-{chunk_index}"); text: str;
    section: str | None; source: str (original filename); title: str (filename stem);
    page_start: int | None; char_start: int; char_end: int
